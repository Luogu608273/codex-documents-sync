#include <bits/stdc++.h>
using namespace std;

using ll = long long;

#define rep(i, a, b) for (int i = (a) ; i <= (b) ; i ++)
#define per(i, a, b) for (int i = (a) ; i >= (b) ; i --)

const int N = 2e5 + 10;

int n, m, col[N];
vector<int> e[N];

int sz[N], son[N], fa[N], dep[N];

void dfs1(int p, int f) {
    sz[p] = 1, fa[p] = f, dep[p] = dep[f] + 1;
    for (auto t : e[p]) {
        if (t == f) continue;
        dfs1(t, p), sz[p] += sz[t];
        if (sz[t] > sz[son[p]]) son[p] = t;
    }
}

int top[N], dfn[N], rdfn[N], pcnt;

void dfs2(int p, int tp) {
    top[p] = tp, dfn[p] = ++ pcnt, rdfn[pcnt] = p;
    if (son[p]) dfs2(son[p], tp);
    for (auto t : e[p])
        if (t != fa[p] && t != son[p]) dfs2(t, t);
}

struct Info {
    int lft, rgt, cnt;
};

Info mrg(Info a, Info b) {
    if (!a.cnt) return b;
    if (!b.cnt) return a;
    return {a.lft, b.rgt, a.cnt + b.cnt - (a.rgt == b.lft)};
}

struct Node {
    int ls, rs, fa;
    int tag;
    Info val;
} tr[N];

void pu(int p) {
    tr[p].val = {col[p], col[p], 1};
    if (tr[p].ls) tr[p].val = mrg(tr[tr[p].ls].val, tr[p].val);
    if (tr[p].rs) tr[p].val = mrg(tr[p].val, tr[tr[p].rs].val);
}

void cover(int p, int c) {
    if (!p) return;
    col[p] = c, tr[p].tag = c;
    tr[p].val = {c, c, 1};
}

void pd(int p) {
    if (tr[p].tag) {
        if (tr[p].ls) cover(tr[p].ls, tr[p].tag);
        if (tr[p].rs) cover(tr[p].rs, tr[p].tag);
        tr[p].tag = 0;
    }
}

int build_chain(int l, int r) {
    if (l > r) return 0;
    int sum = 0;
    rep(i, l, r) sum += sz[rdfn[i]] - sz[son[rdfn[i]]];
    int now = 0, mid = l;
    rep(i, l, r) {
        now += sz[rdfn[i]] - sz[son[rdfn[i]]];
        if (now * 2 >= sum) { mid = i; break; }
    }
    int u = rdfn[mid];
    tr[u].ls = build_chain(l, mid - 1);
    tr[u].rs = build_chain(mid + 1, r);
    if (tr[u].ls) tr[tr[u].ls].fa = u;
    if (tr[u].rs) tr[tr[u].rs].fa = u;
    pu(u);
    return u;
}

int build_gbbt(int p) {
    for (int u = p; u; u = son[u]) {
        for (auto v : e[u]) {
            if (v != fa[u] && v != son[u]) {
                int root = build_gbbt(v);
                tr[root].fa = u;
            }
        }
    }
    int r = p;
    while (son[r]) r = son[r];
    return build_chain(dfn[p], dfn[r]);
}

bool is_root(int p) {
    return !tr[p].fa || (tr[tr[p].fa].ls != p && tr[tr[p].fa].rs != p);
}

void push_all(int p) {
    if (!is_root(p)) push_all(tr[p].fa);
    pd(p);
}

void upd_path(int u, int v, int c) {
    int tu = top[u], tv = top[v];
    while (tu != tv) {
        if (dep[tu] < dep[tv]) swap(u, v), swap(tu, tv);
        push_all(u);
        int cur = u;
        while (cur) {
            col[cur] = c;
            if (tr[cur].ls) cover(tr[cur].ls, c);
            pu(cur);
            if (is_root(cur)) break;
            cur = tr[cur].fa;
        }
        u = fa[tu], tu = top[u];
    }
    if (dep[u] < dep[v]) swap(u, v);
    push_all(u);
    int cur = u;
    while (cur) {
        push_all(cur);
        col[cur] = c;
        if (dfn[cur] >= dfn[v] && tr[cur].ls) cover(tr[cur].ls, c);
        pu(cur);
        if (cur == v || is_root(cur)) break;
        cur = tr[cur].fa;
    }
}

int qry_path(int u, int v) {
    Info lu = {0, 0, 0}, lv = {0, 0, 0};
    while (top[u] != top[v]) {
        if (dep[top[u]] >= dep[top[v]]) {
            push_all(u);
            Info cur_info = {col[u], col[u], 1};
            if (tr[u].ls) cur_info = mrg(tr[tr[u].ls].val, cur_info);
            lu = mrg(cur_info, lu);
            u = fa[top[u]];
        } else {
            push_all(v);
            Info cur_info = {col[v], col[v], 1};
            if (tr[v].ls) cur_info = mrg(tr[tr[v].ls].val, cur_info);
            lv = mrg(cur_info, lv);
            v = fa[top[v]];
        }
    }
    if (dep[u] >= dep[v]) {
        push_all(u);
        Info cur_info = {col[u], col[u], 1};
        if (dfn[u] > dfn[v] && tr[u].ls) cur_info = mrg(tr[tr[u].ls].val, cur_info);
        lu = mrg(cur_info, lu);
    } else {
        push_all(v);
        Info cur_info = {col[v], col[v], 1};
        if (dfn[v] > dfn[u] && tr[v].ls) cur_info = mrg(tr[tr[v].ls].val, cur_info);
        lv = mrg(cur_info, lv);
    }
    swap(lu.lft, lu.rgt);
    return mrg(lu, lv).cnt;
}

void solve() {
    cin >> n >> m;
    rep(i, 1, n) cin >> col[i];
    rep(i, 2, n) {
        int u, v; cin >> u >> v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    dfs1(1, 0);
    dfs2(1, 1);
    build_gbbt(1);

    while (m --) {
        char op; cin >> op;
        if (op == 'C') {
            int u, v, c; cin >> u >> v >> c;
            upd_path(u, v, c);
        } else {
            int u, v; cin >> u >> v;
            cout << qry_path(u, v) << "\n";
        }
    }
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);

    int t = 1;
    while (t--) solve();
}

### 历史助手

这段代码实现的是 **全局平衡二叉树 (Global Balanced Binary Tree, 简称 GBBT)**。

虽然它也使用了树链剖分（HLD）的 `dfs1` 和 `dfs2` 来划分轻重链，但其核心结构和查询方式与传统的“树剖 + 线段树”有显著区别。以下是判断依据：

### 1. 核心构建逻辑 (`build_chain` 和 `build_gbbt`)
*   **权重划分：** 在 `build_chain` 函数中，它并不是简单地取区间中点 `(l+r)/2`，而是计算了 `sz[rdfn[i]] - sz[son[rdfn[i]]]`（即：节点本身加上它所有轻子树的大小之和）。
*   **重心选择：** `if (now * 2 >= sum) { mid = i; break; }` 这行代码是为了寻找该重链的**权重重心**。这是 GBBT 的标准写法，目的是为了让树的高度保持在 $O(\log n)$，即使在处理轻重链切换时也能保证全局深度平衡。
*   **树形结构：** 它使用了类似 LCT（Link-Cut Tree）的结构，每个节点有 `ls`（左儿子）、`rs`（右儿子）和 `fa`（父亲）。这是一种**静态 LCT** 的实现方式。

### 2. 结构特征
*   **线段树：** 传统的树剖线段树是建立在 `dfn` 序上的一个（或多个）完全二叉树，通常使用 `p << 1` 和 `p << 1 | 1` 来访问子节点。
*   **GBBT：** 它是为每一条重链构建一棵平衡二叉树，并把轻儿子的重链根节点连到父节点上。代码中的 `is_root` 函数 `(tr[tr[p].fa].ls != p && tr[tr[p].fa].rs != p)` 是典型的辅助树根节点判断（与 LCT 一致），用于区分当前节点是否为所在重链平衡树的根。

### 3. 查询与修改方式
*   代码中的 `upd_path` 和 `qry_path` 结合了 HLD 的跳链（`top[u]`）和 GBBT 的树状遍历。
*   由于 GBBT 的性质，它可以在不使用线段树的情况下，通过维护平衡树的信息来完成路径操作。虽然这份代码在跳链时仍然使用了 `top` 数组，但它内部处理重链区间时是沿着 `fa` 指针向上爬平衡树或直接访问平衡树子树。

### 总结
*   **树剖线段树**：$O(\log^2 n)$，基于 `dfn` 序的数组下标维护。
*   **全局平衡二叉树 (GBBT)**：$O(\log n)$，基于轻重链权重平衡构建的二叉树。

**结论：** 这份代码是 **全局平衡二叉树 (GBBT)**。它通过这种方式将树剖原本的 $O(\log^2 n)$ 操作在理论或实践上优化到了更紧凑的结构（虽然某些实现下跳链仍为 $\log^2$，但由于其树高的平衡性，性能通常优于普通树剖线段树）。

### 历史助手

全局平衡二叉树（GBBT）的时间复杂度之所以能做到 **全局树高为 $O(\log n)$**（从而使得单次路径操作达到 $O(\log n)$），关键在于它巧妙地结合了**树链剖分（HLD）**和**带权二分（Weighted Divide and Conquer）**。

其核心证明逻辑可以总结为一句话：**在 GBBT 中，从任意节点沿父节点指针向上走一步，其代表的“子树节点总数（权重）”至少翻倍。**

下面是具体的数学原理拆解：

---

### 1. 定义“节点权重” $W(u)$

在 GBBT 中，我们将一个节点 $u$ 的**子树大小**（包含其在 GBBT 中所有子孙节点对应的原树节点数量之和）定义为 $W(u)$。
* 对于 GBBT 的根节点，其权重 $W(\text{root}) = n$（原树的总节点数）。
* 对于 GBBT 的叶子节点，其权重 $W(\text{leaf}) \ge 1$。

如果能证明：**从任何节点 $v$ 跳到它的父节点 $u$（即 $u = \text{fa}[v]$），都满足 $W(u) \ge 2 \cdot W(v)$**，那么从叶子节点到根节点最多只需要跳 $\log_2 n$ 次，树高即为 $O(\log n)$。

---

### 2. GBBT 的两条向上边（推导过程）

GBBT 中的边分为两类：**重链内部的 BST 边** 和 **重链之间的轻边**。我们分别证明这两类边上的翻倍性质。

#### 边类型 A：重链内部的 BST 边（内边）

在为某条重链构建二叉搜索树（BST）时，我们使用的是**带权二分**：
1. 重链上每个点 $x$ 的个人权重为 $w(x) = 1 + \sum \text{sz}(\text{轻儿子})$。
2. 在对区间 $[L, R]$ 建树时，选择一个点 $mid$，使得：
   * 左子树的权重和 $\le \frac{1}{2} \times \text{当前区间的总权重和}$
   * 右子树的权重和 $\le \frac{1}{2} \times \text{当前区间的总权重和}$

**推论：**
当我们在重链内部的 BST 中从子节点 $v$ 走向父节点 $u$ 时，由于 $v$ 是通过二分划分出来的子区间，**子节点 $v$ 的权重 $W(v)$ 必定不超过父节点区间总权重 $W(u)$ 的一半**。
即：
$$W(v) \le \frac{1}{2} W(u) \implies W(u) \ge 2 \cdot W(v)$$
**结论：在重链 BST 内部每向上走一步，权重 $W$ 至少翻倍。**

---

#### 边类型 B：重链与重链之间的轻边（跨链边）

假设节点 $r$ 是某条轻链在 GBBT 中的根节点，它的父节点是主链上的节点 $p$（在原树中，$r$ 是 $p$ 的一个轻儿子）。

1. 根据树链剖分（HLD）的性质：
   轻儿子的子树大小不超过父节点子树大小的一半，即：
   $$\text{sz}(r) \le \frac{1}{2} \text{sz}(p)$$
2. 在 GBBT 中：
   * $r$ 作为轻链的根，其在 GBBT 中的权重 $W(r) = \text{sz}(r)$。
   * $p$ 所在重链的 BST 根节点 $R_p$，其权重 $W(R_p) = \text{sz}(\text{top}(p)) \ge \text{sz}(p)$。

因此，从轻链根 $r$ 跳到父重链的 BST 根节点 $R_p$，权重满足：
$$W(R_p) \ge \text{sz}(p) \ge 2 \cdot \text{sz}(r) = 2 \cdot W(r)$$

**结论：从一条轻链的根节点跳到上一级重链的根节点，权重 $W$ 也至少翻倍。**

---

### 3. 总体结论

无论是：
1. **在重链内部向上走**（利用带权二分保证翻倍）；
2. **跨越重链向上跳**（利用树剖轻重边性质保证翻倍）；

**每一次有效的向上攀升，对应子树的节点规模 $W$ 都至少乘以 2。**

因为全局节点总数只有 $n$，权重从 $\ge 1$ 增长到 $n$ 最多只需翻倍 $\log_2 n$ 次。

这就是为什么全局平衡二叉树的**整棵树的高度严格限制在 $O(\log n)$**。

由于树高是 $O(\log n)$，在路径修改/查询时，只需像使用 Splay/LCT 一样直接沿父节点指针向上 pushup/pushdown，或者直接在 GBBT 上找 LCA 并合并信息，即可在 **$O(\log n)$** 时间内完成操作，消去了传统树剖线段树额外的一层 $\log$（即从 $O(\log^2 n)$ 优化到了 $O(\log n)$）。
