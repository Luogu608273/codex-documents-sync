#include <bits/stdc++.h>
#define endl '\n'

using namespace std;

using ll = long long;

const ll N = 1e6 + 5;

int n, m, p[N], dep[N], col[N], f[N], siz[N], tp[N];
map<int, vector<int>> mp[N];

int find(int x) { return f[x] == x ? x : f[x] = find(f[x]); }

int merge(int u, int v)
{
	u = find(u), v = find(v);
	if (u == v)
		return u;
	if (mp[u].size() < mp[v].size())
		swap(u, v);
	f[v] = u;
	siz[u] += siz[v];
	if (dep[tp[v]] < dep[tp[u]])
		tp[u] = tp[v];
	for (auto &it : mp[v])
	{
		if (mp[u][it.first].size() < it.second.size())
			swap(mp[u][it.first], it.second);
		mp[u][it.first].insert(mp[u][it.first].end(), it.second.begin(), it.second.end());
	}
	mp[v].clear();
	return u;
}

void work(int rt, int y)
{
	while (1)
	{
		rt = find(rt);
		auto it = mp[rt].find(y);
		if (it == mp[rt].end() || it->second.empty())
			break;
		int ch = find(it->second.back());
		it->second.pop_back();
		if (ch != rt && col[ch] == y && find(p[tp[ch]]) == rt)
			rt = merge(rt, ch);
	}
}

void init() {}

void solve()
{
	cin >> n >> m;
	dep[1] = 1;
	for (int i = 2; i <= n; ++i)
	{
		cin >> p[i];
		dep[i] = dep[p[i]] + 1;
	}
	for (int i = 1; i <= n; ++i)
	{
		cin >> col[i];
		f[i] = tp[i] = i;
		siz[i] = 1;
	}
	for (int i = 2; i <= n; ++i)
	{
		if (col[i] == col[p[i]])
			merge(i, p[i]);
	}
	for (int i = 2, rx; i <= n; ++i)
	{
		rx = find(i);
		if (tp[rx] == i)
		{
			int pr = find(p[i]);
			mp[pr][col[rx]].push_back(rx);
		}
	}
	for (int opt, x, y, rx, rt; m--;)
	{
		cin >> opt;
		if (opt == 1)
		{
			cin >> x >> y;
			rx = find(x);
			if (col[rx] == y)
				continue;
			col[rx] = y;
			if (p[tp[rx]])
			{
				rt = find(p[tp[rx]]);
				if (col[rt] == y)
					rx = merge(rx, rt);
				else
					mp[rt][y].push_back(rx);
			}
			work(rx, y);
		}
		else
		{
			cin >> x;
			cout << siz[find(x)] << endl;
		}
	}
}

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cout.tie(nullptr);
	int _ = 1;
	// cin >> _;
	init();
	while (_--)
		solve();
	return 0;
}
# P9067 [Ynoi Easy Round 2022] 虚空处刑 TEST_105

## 题目描述

给定一棵 $n$ 个节点的树，第 $i$ 个点有点权 $a_i$。

定义一个点 $x$ 所在的极大同色连通块为一个**极大的**点集 $S$，满足 $x\in S$，且对任意点集中的元素 $i,j$，可以找到一个节点序列 $p_1,p_2,...p_t$，满足 $p_1=i$，$p_t=j$，且对任意 $k$ 为 $[1,t)$ 中的整数，满足 $p_k$ 与 $p_{k+1}$ 在树上相邻，且 $a_{p_k}=a_{p_{k+1}}$，且 $p_k\in S$。

有 $m$ 次操作：

`1 x y`：给出一个点 $x$ ，将其所在的极大同色连通块中每个点的点权修改为 $y$。

`2 x`：给出一个点 $x$，查询其所在的极大同色连通块的大小。

## 输入格式

第一行两个数 $n,m$。

第二行 $n-1$ 个数，第 $i$ 个数表示树上第 $i+1$ 的节点的父亲节点的编号，保证父亲节点的编号比该节点小。

第三行 $n$ 个数，第 $i$ 个数表示 $a_i$。

之后 $m$ 行，每行形如 `1 x y` 或 `2 x`，意义如上述。

## 输出格式

对每个 $2$ 操作，输出一行一个数表示答案。

## 输入输出样例 #1

### 输入 #1

```
4 5
1 1 2
3 1 1 1
2 4
1 1 1
1 4 3
2 4
1 3 3
```

### 输出 #1

```
2
4
```

## 说明/提示

对于 $20\%$ 的数据，满足 $n,m\leq2\times 10^3$。

对于 $40\%$ 的数据，满足 $n,m\leq2\times 10^5$。

对于另外 $30\%$ 的数据，满足 $1\le a_i,y\le 2$。

对于 $100\%$ 的数据，满足 $1\le n,m,a_i,x,y\le10^6$。

### 历史助手

以下是使用 **01Trie + Trie 合并** 解决本题的 C++ 代码。

### 解题思路
1. **树上同色连通块维护**：由于树的结构固定且以 $1$ 为根，每个同色连通块对应树上的一个连通子图，其深度最小的点（即最高节点 $tp$）唯一确定该连通块。
2. **只维护向下子连通块**：在连通块内，不主动记录父节点方向的信息，而是为每个连通块的最高节点维护一棵 01Trie。Trie 中以**子连通块的颜色**为 key，挂载所有以当前连通块内的点为父亲的**子连通块最高节点**。
3. **启发式合并 / Trie 合并**：
   - 当连通块颜色修改为 $y$ 时，首先检查其父节点连通块颜色是否为 $y$，若是则合并。
   - 接着从 Trie 中取出所有颜色为 $y$ 的子连通块，逐一进行 DSU 合并，同时调用 01Trie 的合并算法（Trie Merge）。
   - 利用懒标记/条件校验思想：提取 Trie 节点时，若发现已被合并或颜色不符，直接忽略。

```cpp
#include <bits/stdc++.h>
#define endl '\n'

using namespace std;

const int N = 1000005;
const int T = 15000005;

int n, m, p[N], dep[N], col[N], f[N], siz[N], tp[N], rt[N];
int ch[T][2], rub[T], top_rub, tot_t;

struct Node {
	int id, nxt;
} e[N * 2];
int tot_e;

int find(int x) {
	return f[x] == x ? x : f[x] = find(f[x]);
}

int new_node() {
	int u = top_rub ? rub[top_rub--] : ++tot_t;
	ch[u][0] = ch[u][1] = 0;
	return u;
}

void del_node(int u) {
	ch[u][0] = ch[u][1] = 0;
	rub[++top_rub] = u;
}

void insert(int &r, int val, int id) {
	if (!r) r = new_node();
	int u = r;
	for (int i = 19; i >= 0; --i) {
		int b = (val >> i) & 1;
		if (!ch[u][b]) ch[u][b] = new_node();
		u = ch[u][b];
	}
	e[++tot_e] = {id, 0};
	if (!ch[u][0]) ch[u][0] = ch[u][1] = tot_e;
	else {
		e[ch[u][1]].nxt = tot_e;
		ch[u][1] = tot_e;
	}
}

int merge_trie(int u, int v, int b) {
	if (!u || !v) return u ? u : v;
	if (b < 0) {
		if (!ch[u][0]) {
			ch[u][0] = ch[v][0];
			ch[u][1] = ch[v][1];
		} else if (ch[v][0]) {
			e[ch[u][1]].nxt = ch[v][0];
			ch[u][1] = ch[v][1];
		}
		del_node(v);
		return u;
	}
	ch[u][0] = merge_trie(ch[u][0], ch[v][0], b - 1);
	ch[u][1] = merge_trie(ch[u][1], ch[v][1], b - 1);
	del_node(v);
	return u;
}

int merge_comp(int u, int v) {
	u = find(u); v = find(v);
	if (u == v) return u;
	if (dep[tp[u]] > dep[tp[v]]) swap(u, v);
	f[v] = u;
	siz[u] += siz[v];
	rt[u] = merge_trie(rt[u], rt[v], 19);
	return u;
}

int get_leaf(int r, int val) {
	if (!r) return 0;
	int u = r;
	for (int i = 19; i >= 0; --i) {
		int b = (val >> i) & 1;
		u = ch[u][b];
		if (!u) return 0;
	}
	return u;
}

void process_children(int u, int y) {
	while (1) {
		u = find(u);
		int leaf = get_leaf(rt[u], y);
		if (!leaf || !ch[leaf][0]) break;
		int list_head = ch[leaf][0];
		ch[leaf][0] = ch[leaf][1] = 0;
		int cur = list_head;
		while (cur) {
			int id = e[cur].id;
			int nxt_e = e[cur].nxt;
			int cv = find(id);
			if (cv != u && col[cv] == y && tp[cv] == id && find(p[id]) == u) {
				u = merge_comp(u, cv);
			}
			cur = nxt_e;
		}
	}
}

void change_color(int x, int y) {
	int u = find(x);
	if (col[u] == y) return;
	col[u] = y;
	int T = tp[u];
	if (T != 1) {
		int P = find(p[T]);
		if (col[P] == y) {
			u = merge_comp(u, P);
		} else {
			insert(rt[P], y, T);
		}
	}
	process_children(u, y);
}

void solve() {
	cin >> n >> m;
	dep[1] = 1;
	for (int i = 2; i <= n; ++i) {
		cin >> p[i];
		dep[i] = dep[p[i]] + 1;
	}
	for (int i = 1; i <= n; ++i) {
		cin >> col[i];
		f[i] = tp[i] = i;
		siz[i] = 1;
	}
	for (int i = 2; i <= n; ++i) {
		if (col[i] == col[p[i]]) merge_comp(i, p[i]);
	}
	for (int i = 2; i <= n; ++i) {
		int r = find(i);
		if (tp[r] == i) {
			int P = find(p[i]);
			insert(rt[P], col[r], i);
		}
	}
	for (int opt, x, y; m--;) {
		cin >> opt;
		if (opt == 1) {
			cin >> x >> y;
			change_color(x, y);
		} else {
			cin >> x;
			cout << siz[find(x)] << endl;
		}
	}
}

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cout.tie(nullptr);
	solve();
	return 0;
}
```
